\documentclass[tikz]{standalone}
\usepackage{tikz}
\definecolor{bgdark}{HTML}{171D2B}
\definecolor{acc}{HTML}{F7B538}
\begin{document}
\begin{tikzpicture}[text=white, x=0.85cm, y=0.85cm]
\fill[bgdark] (-1.2,-1.6) rectangle (17.6,5.4);
% empat segitiga bilangan: n = 1,2,3,4
\foreach \n/\x0 in {1/0.5, 2/3.9, 3/7.9, 4/12.4} {
  \foreach \r in {1,...,\n} {           % baris ke-r dari bawah punya r titik
    \foreach \c in {1,...,\r}
      \fill[acc] (\x0+\c-1, \n-\r) circle (0.14);
  }
  \node[below=0.5cm] at (\x0+\n/2-0.5, -0.1) {$T_{\n}=\pgfmathtruncatemacro{\s}{\n*(\n+1)/2}\s$};
  \node[above=0.3cm] at (\x0+\n/2-0.5, \n*0.85) {titik ke-\n};
}
% panah selisih antar segitiga
\foreach \a/\b/\n in {0.5/3.9/1, 3.9/7.9/2, 7.9/12.4/3} {
  \draw[gray!70, -{Stealth}] (\a+\n*0.85+0.9, 2.6) -- (\b-0.6, 2.6)
     node[midway, above, font=\small] {$+\n$};
}
